def nearcol(s,target,tol=50,opaque=True):
 nums=[float(v) for v in re.findall(r'[\d.]+',s or '')]
 return len(nums)>=3 and (not opaque or len(nums)<4 or nums[3]>.9) and math.dist(nums[:3],target)<=tol
def ordered(rows):return all(r.get('row') for r in rows) and [r['name'] for r in sorted(rows,key=lambda x:x['row']['y'])]
